	# =========================================================================
	# RAMA 1: MINISTRO DE ECONOMÍA Y PLANIFICACIÓN INDUSTRIAL (AUTÓNOMO)
	# =========================================================================
	gemini_autopilot_cycle = {
		icon = generic_industry
		cost = 0
		days_remove = 30
		visible = {
			NOT = { has_country_flag = gemini_autopilot_disabled }
		}
		available = {
			always = yes
		}
		remove_effect = {
			# Disparar Evento Presidencial si ya tienes 35 Civiles o es 1938
			if = {
				limit = {
					num_of_civilian_factories > 34
					NOT = { has_country_flag = gemini_minister_event_1_fired }
				}
				set_country_flag = gemini_minister_event_1_fired
				country_event = { id = gemini_minister.1 days = 1 }
			}

			# Construcción Automática mensual inteligente
			if = {
				limit = { has_country_flag = gemini_focus_mils }
				random_owned_controlled_state = {
					limit = { free_building_slots = { building = arms_factory size > 0 } }
					add_building_construction = { type = arms_factory level = 1 instant_build = no }
				}
			}
			else = {
				random_owned_controlled_state = {
					limit = { free_building_slots = { building = industrial_complex size > 0 } }
					add_building_construction = { type = industrial_complex level = 1 instant_build = no }
				}
			}
		}
		ai_will_do = { factor = 0 }
	}
